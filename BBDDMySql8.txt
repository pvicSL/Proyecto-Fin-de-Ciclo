CREATE DATABASE ESTUDIO_TATUAJE;
USE ESTUDIO_TATUAJE;

CREATE TABLE IF NOT EXISTS CLIENTES (
id_cliente INTEGER PRIMARY KEY AUTO_INCREMENT,
nombre VARCHAR (30) NOT NULL,
apellido1 VARCHAR (30) NOT NULL,
apellido2 VARCHAR (30),
email VARCHAR (50) NOT NULL,
telefono VARCHAR (30) NOT NULL,
UNIQUE INDEX ID_UNIQUE_CLIENTE (nombre, apellido1, telefono) /*Evita inserción de registros duplicados*/
);

CREATE TABLE IF NOT EXISTS TRABAJADORES (
id_trabajador INT PRIMARY KEY AUTO_INCREMENT,
contrasenia INTEGER (30) NOT NULL,
nombre VARCHAR (30) NOT NULL,
apellido1 VARCHAR (30) NOT NULL,
apellido2 VARCHAR (30),
email VARCHAR (50) NOT NULL,
telefono VARCHAR (30) NOT NULL,
rol ENUM ('ADMIN', 'TRABAJADOR') NOT NULL,	/*controla las opciones del rol*/
funciones ENUM ('CREACION', 'ELIMINACION') NOT NULL
);

CREATE TABLE IF NOT EXISTS SERVICIOS (
id_servicio INT PRIMARY KEY AUTO_INCREMENT,
tipo ENUM ('TATUAJE', 'ELIMINACION', 'COVER', 'RETOQUE'),
zona ENUM ('BRAZO', 'ANTEBRAZO', 'CODO', 'HOMBRO', 'TÓRAX', 'ABDOMEN', 'PUBIS', 'MUSLO', 'RODILLA', 'PANTORILLA', 'PIE', 'MANO', 'CERVIAL', 'LUMBARES', 'NALGA', 'CABEZA'),
tamanio ENUM ('MINI', 'PEQUEÑO', 'MEDIANO', 'GRANDE', 'MUY_GRANDE'),
detalle ENUM ('SENCILLO', 'MEDIO', 'DENSO'),
estilo ENUM ('ANIME', 'BLACKANDGREY', 'BLACK', 'FINELINE', 'FULLCOLOR', 'LETERING', 'JAPONES', 'REALISMO', 'TRADICIONAL'),
comentarios VARCHAR (200),
id_cliente INTEGER,
FOREIGN KEY (id_cliente) REFERENCES CLIENTES (id_cliente)
/*PENDIENTE RUTAS IMÁGENES*/
);
